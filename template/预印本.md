function richTables(text,parseYaml,stringifyYaml){
 const dump=data=>stringifyYaml(data).trimEnd();
 let blocks=[],tables=[];let s=text.replace(/```[\s\S]*?```|~~~[\s\S]*?~~~/g,b=>{if(/^```rich-table/.test(b)){let x=b.match(/^```rich-table[^\n]*\n([\s\S]*?)```/);const payload=x[1].replace(/^---\r?\n/,'').replace(/\r?\n---\s*$/,'');let data;try{data=JSON.parse(payload);}catch{if(typeof parseYaml!=='function')throw Error('Obsidian parseYaml is required for existing Rich Table YAML');data=parseYaml(payload);}if(!data||typeof data!=='object')throw Error('Invalid Rich Table data');if(data.xlsxSource||data.sheets?.some(sheet=>sheet.xlsxSource))throw Error('外部xlsx表的锁定语义不同，停止并核对，不能声称图形编辑已锁定');if(data.version===3){if(!Array.isArray(data.sheets)||!data.sheets.length)throw Error('Workbook sheets missing');for(const sheet of data.sheets){sheet.locked=true;sheet.theme='grid';}}else if(data.version===2){data.locked=true;data.theme='grid';}else throw Error('Unsupported Rich Table version; preserve and review rather than overwrite');return '```rich-table\n---\n'+dump(data)+'\n---\n```';}return '\u0000BLOCK'+(blocks.push(b)-1)+'\u0000';});
 s=s.replace(/^\|[^\r\n]+\|[ \t]*\r?\n\|[ :|\-]+\|[ \t]*(?:\r?\n\|[^\r\n]+\|[ \t]*)+/gm,raw=>{
  const lines=raw.trimEnd().split(/\r?\n/),split=line=>{let a=[],v='',math=false,code=false;const t=line.trim().slice(1,-1);for(let i=0;i<t.length;i++){const ch=t[i];if(ch==='\\'&&t[i+1]==='|'){v+='\\|';i++;continue;}if(ch==='`')code=!code;if(ch==='$'&&!code&&t[i-1]!=='\\')math=!math;if(ch==='|'&&!math&&!code){a.push(v.trim());v='';}else v+=ch;}a.push(v.trim());return a;};const cells=lines.filter((l,i)=>i!==1).map(split),width=cells[0].length;if(cells.some(r=>r.length!==width))throw Error('Ambiguous Markdown table width');
  const columns=cells[0].map((name,i)=>({id:'c_'+String(i+1).padStart(6,'0'),name}));const data={version:2,columns,rows:cells.slice(1).map((r,i)=>({id:'r_'+String(i+1).padStart(6,'0'),cells:Object.fromEntries(r.map((v,j)=>[columns[j].id,v]))})),theme:'grid',locked:true};
  const block='```rich-table\n---\n'+dump(data)+'\n---\n```';tables.push({raw,block,rows:cells.length-1,columns:width});return block;
 });
 return {text:s.replace(/\u0000BLOCK(\d+)\u0000/g,(_,i)=>blocks[+i]),tables};
}
function findBibliography(body){
 const pattern=/^ {0,3}(`{3,}|~{3,})[^\n]*$|^#{1,6}[ \t]+(?:\d+(?:\.\d+)*[ \t]+)?(?:参考文献|References\b|Bibliography\b)[^\n]*$/gmi;
 let fence=null;
 for(const match of body.matchAll(pattern)){
  const marker=match[1];
  if(marker){
   if(!fence)fence=marker;
   else if(marker[0]===fence[0]&&marker.length>=fence.length&&!match[0].trim().slice(marker.length).trim())fence=null;
  }else if(!fence)return match;
 }
 return null;
}
function prepareBody(body,profile){
 const pagePattern=/[〔【\[（(]\s*(?:(?:源|来源|原文|source)\s*(?:页码|页|pages?)?\s*[:：]?\s*)?(?:p{1,2}\.?\s*\d+(?:\s*[-–—]\s*p?\.?\s*\d+)?|第?\s*\d+(?:\s*[-–—]\s*\d+)?\s*页)\s*[〕】\]）)]/gi;
 const allowed=new Set(profile.originalPageReferences||[]),unreviewed=[...body.matchAll(pagePattern)].map(m=>m[0]).filter(s=>!allowed.has(s));
 if(unreviewed.length)throw Error('核对并移除人为来源页码；原文本身有意义的页码在 originalPageReferences 中准确列出以保留：'+[...new Set(unreviewed)].join('；'));
 const mappings=profile.citationLinks||[];if(!Array.isArray(mappings))throw Error('citationLinks must be an array');
 if(!mappings.length)return body;
 const header=findBibliography(body);if(!header)throw Error('引用映射须对应已核实的参考文献部分');
 let prose=body.slice(0,header.index),references=body.slice(header.index),entries=new Map();
 for(const m of references.matchAll(/^(\d+)\. ([^\n]+)$/gm)){const number=+m[1];if(entries.has(number))throw Error('参考文献编号重复');entries.set(number,m[0]);}
 const protectedSpans=[...prose.matchAll(/```[\s\S]*?```|\$\$[\s\S]*?\$\$|\[\[[^\n]+?\]\]|(?<!!)\[(?:\\.|[^\]\\])+\]\([^\n]*?\)/g)].map(m=>[m.index,m.index+m[0].length]);
 const edits=[],used=new Set();
 for(const mapping of mappings){const {text,reference,occurrence}=mapping;if(typeof text==='string'&&/^\s*(?:\[\s*)?\d+(?:[\s,，;；–—−-]+\d+)*\s*\]?\s*$/.test(text))throw Error('数字引文请在输入稿中完成整组方括号、上标和书目链接');if(typeof text!=='string'||!text||!Number.isInteger(reference)||!entries.has(reference)||!Number.isInteger(occurrence)||occurrence<1)throw Error('引用映射须给出准确 text、reference 和 occurrence，不按相似标题猜配');
  const positions=[];let offset=0;while((offset=prose.indexOf(text,offset))>=0){if(!protectedSpans.some(([a,b])=>offset<b&&offset+text.length>a))positions.push(offset);offset+=text.length;}
  const index=positions[occurrence-1];if(index===undefined)throw Error('引用文字及出现次序不匹配：'+text);
  if(edits.some(e=>index<e.end&&index+text.length>e.start))throw Error('引用映射重叠');
  const label=text.replace(/\\/g,'\\\\').replace(/[\[\]]/g,'\\$&');edits.push({start:index,end:index+text.length,value:'['+label+'](<#^ref-'+reference+'>)'});used.add(reference);
 }
 for(const e of edits.sort((a,b)=>b.start-a.start))prose=prose.slice(0,e.start)+e.value+prose.slice(e.end);
 for(const number of used){const line=entries.get(number),id='ref-'+number,occurrences=[...body.matchAll(new RegExp('\\^'+id+'(?=\\s|$)','g'))];if(occurrences.length&&!line.endsWith(' ^'+id))throw Error('已有同名引用块编号，先核对，不能覆盖：'+id);if(occurrences.length>1)throw Error('引用块编号不唯一');if(!line.endsWith(' ^'+id))references=references.replace(line,line+' ^'+id);}
 return prose+references;
}
function prepareFigureLinks(body,profile){
 const targets=profile.figureTargets||[],mappings=profile.figureLinks||[];
 if(!Array.isArray(targets)||!Array.isArray(mappings))throw Error('figureTargets / figureLinks must be arrays');
 if(!targets.length&&!mappings.length)return body;
 const bibliography=findBibliography(body),proseEnd=bibliography?bibliography.index:body.length;
 const protectedSpans=[...body.slice(0,proseEnd).matchAll(/```[\s\S]*?```|\$\$[\s\S]*?\$\$|\[\[[^\n]+?\]\]|(?<!!)\[(?:\\.|[^\]\\])+\]\([^\n]*?\)|^\|[^\n]+\|[ \t]*\n\|[ :|\-]+\|[ \t]*(?:\n\|[^\n]+\|[ \t]*)+/gm)].map(m=>[m.index,m.index+m[0].length]);
 protectedSpans.push([proseEnd,body.length]);
 const edits=[],numbers=new Map(),ids=new Set();
 for(const target of targets){const {number,image,caption}=target;
  if(typeof number!=='string'||!/^(?:[A-Za-z]\.?)*\d+(?:\.\d+)*$/.test(number)||typeof image!=='string'||!image.startsWith('![')||image.includes('\n')||typeof caption!=='string'||!caption)throw Error('图目标须给出经原文核实的编号、完整图片嵌入行和完整题注');
  if(numbers.has(number)||body.split(image).length!==2||body.split(caption).length!==2)throw Error('图号/图片/题注不唯一，停止核对');
  const id='fig-'+number.toLowerCase().replaceAll('.','-');if(ids.has(id))throw Error('图锚点命名冲突');ids.add(id);
  const start=body.indexOf(image),end=start+image.length,captionStart=body.indexOf(caption),captionEnd=captionStart+caption.length;
  const titleNumber=(caption.replace(/^\*\*/,'').match(/^(?:图|Figure|Fig\.)\s*([A-Za-z]?(?:\.\s*)?\d+(?:\.\d+)*)(?=\s|[.。：:（(、，]|[\u3400-\u9fff]|$)/i)||[])[1]?.replace(/\s/g,'');
  if(titleNumber!==number||start>=proseEnd||captionStart<end||captionEnd>proseEnd)throw Error('图号与真实题注/位置不一致');
   const gap=body.slice(end,captionStart),anchor='^'+id;

   const gapLines=gap.split('\n').map(line=>line.trim()).filter(Boolean);
   const afterCaption=existing[0]?.index===captionEnd+1&&body[captionEnd]==='\n';
   const between=existing.length===1&&existing[0].index>=end&&existing[0].index<captionStart&&gapLines.length===1&&gapLines[0]===anchor;
   if(existing.length>1||(existing.length&&!afterCaption&&!between))throw Error('图块锚点冲突');
   if(!gap.includes('\n')||(gapLines.length&&!between))throw Error('题注须位于图片下一行');
   if(gap!=='  \n')edits.push({start:end,end:captionStart,value:'  \n'});
   if(!afterCaption)edits.push({start:captionEnd,end:captionEnd,value:'\n'+anchor});
  protectedSpans.push([start,captionEnd]);numbers.set(number,{id,image,caption});
 }
 for(const mapping of mappings){const {text,number,occurrence,context}=mapping,target=numbers.get(number);
  if(!target||typeof text!=='string'||!text||!Number.isInteger(occurrence)||occurrence<1)throw Error('正文图引用须提供原显示文字、已核实图号和出现次序');
  if(new Set(text.match(/\d+(?:\.\d+)*/g)||[]).size>1)throw Error('跨图组合/范围不得伪配单个图目标；按原文逐个显式编号链接，无法可靠拆分则保留并报告');
  let low=0,high=proseEnd;
  if(context!==undefined){if(typeof context!=='string'||!context||body.split(context).length!==2)throw Error('组合图号的原文上下文须准确且唯一');low=body.indexOf(context);high=low+context.length;}
  const positions=[];let offset=low;
  while((offset=body.indexOf(text,offset))>=0&&offset+text.length<=high){if(!protectedSpans.some(([a,b])=>offset<b&&offset+text.length>a))positions.push(offset);offset+=text.length;}
  const index=positions[occurrence-1];if(index===undefined)throw Error('图引用不存在，或位于题注、图片、已有链接、代码/表格或参考文献内');
  if(edits.some(e=>index<e.end&&index+text.length>e.start))throw Error('图引用映射重叠');
  const label=text.replace(/\\/g,'\\\\').replace(/[\[\]]/g,'\\$&');edits.push({start:index,end:index+text.length,value:'['+label+'](<#^'+target.id+'>)'});
 }
 for(const edit of edits.sort((a,b)=>b.start-a.start))body=body.slice(0,edit.start)+edit.value+body.slice(edit.end);
 return body;
}
function prepareTableLinks(body,profile,parseYaml){

 for(const m of body.slice(0,proseEnd).matchAll(/^!\[[^\n]+\n(?:\^[^\n]+\n)?\n(?:\*\*)?(?:图|Figure|Fig\.)[^\n]+/gm))protectedSpans.push([m.index,m.index+m[0].length]);
  const edits=[],numbers=new Map(),registered=new Set(),anchorIds=new Set();
  const fences=/^ {0,3}(`{3,}|~{3,})([^\n]*)\n([\s\S]*?)^ {0,3}\1[ \t]*(?=\n|$)/gm;
  const definitions=[...body.replace(fences,'').matchAll(/(?:^|[ \t])\^(tbl-[A-Za-z0-9-]+)[ \t]*(?=\n|$)/gm)].map(m=>m[1]);
  const tables=[...body.slice(0,proseEnd).matchAll(fences)]

   .map(m=>({start:m.index,end:m.index+m[0].length,raw:m[0],data:parseYaml(m[3].replace(/^---\n/,'').replace(/\n---\s*$/,'').trim())}));

   if(after){table.anchor=after[1];table.anchorEnd=table.end+after[0].length;}
   protectedSpans.push([table.start,table.anchorEnd??table.end]);
  }

  });

   if(typeof number!=='string'||typeof caption!=='string'||caption.includes('\n')||tableLabel(caption)!==number||numbers.has(number))throw Error('表目标须给出经原文核实的唯一编号与完整题注');

   }).filter(({table,preceding})=>table.data?.title===caption||preceding);

   }
   if(!table.data.title){

    const formatted='**表'+number+'：** '+caption.replace(labelPrefix,'');
    edits.push({start:table.start,end:table.end,value:table.raw.replace(/^---\n/m,mark=>mark+'title: '+JSON.stringify(formatted)+'\n')});
   }
  }
  for(const table of tables){

   anchorIds.add(id);

   const following=body.slice(end).match(/^\n(?:[ \t]*\n)*/)?.[0]??'';
   const spacing=end+following.length<body.length?'\n\n':'';

  }
 for(const mapping of mappings){const {text,number,occurrence,context}=mapping,id=numbers.get(number);
  if(!id||typeof text!=='string'||!text||!Number.isInteger(occurrence)||occurrence<1)throw Error('正文表引用须提供原显示文字、已核实表号和出现次序');
  if(new Set(text.match(/\d+(?:\.\d+)*/g)||[]).size>1)throw Error('跨表组合/范围不得伪配单个表目标；逐个显式编号链接，不可靠时保留并报告');
  let low=0,high=proseEnd;if(context!==undefined){if(typeof context!=='string'||!context||body.split(context).length!==2)throw Error('组合表号的原文上下文须准确且唯一');low=body.indexOf(context);high=low+context.length;}
  const positions=[];let offset=low;while((offset=body.indexOf(text,offset))>=0&&offset+text.length<=high){if(!protectedSpans.some(([a,b])=>offset<b&&offset+text.length>a))positions.push(offset);offset+=text.length;}
  const index=positions[occurrence-1];if(index===undefined)throw Error('表引用不存在，或位于题注、表内容、图题注、已有链接、参考文献/排除段内');
  if(edits.some(e=>index<e.end&&index+text.length>e.start))throw Error('表引用映射重叠');
  const label=text.replace(/\\/g,'\\\\').replace(/[\[\]]/g,'\\$&');edits.push({start:index,end:index+text.length,value:'['+label+'](<#^'+id+'>)'});
 }
 for(const edit of edits.sort((a,b)=>b.start-a.start))body=body.slice(0,edit.start)+edit.value+body.slice(edit.end);
 return body;
}
function prepareEquationLinks(body,profile){

 if(!Array.isArray(targets)||!Array.isArray(mappings)||!Array.isArray(excluded))throw Error('equationTargets / equationLinks / equationExcludedContexts must be arrays');

 const edits=[],numbers=new Set();let index=0;
 for(const line of body.split('\n')){
  const start=index,end=start+line.length;index=end+1;
  if(start>=proseEnd||codeSpans.some(([a,b])=>start>=a&&start<b)||blocks.some(m=>start>=m.index&&start<m.index+m[0].length))continue;

  const adjacent=blocks.filter(m=>{const mathEnd=m.index+m[0].length;if(mathEnd>=start)return false;return /^\s*(?:\^eq-[a-z0-9-]+\s*)?$/.test(body.slice(mathEnd,start));});

  const block=adjacent[0],tags=[...block[0].matchAll(/\\tag\*?\{([^}]+)\}/g)];if(tags.length>1||(tags.length===1&&tags[0][1]!==number))throw Error('已有公式tag与外部编号不一致，停止核对');
